cmake_minimum_required(VERSION 3.5)
project(jetson_ptz_inspection CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O2 ${OpenMP_CXX_FLAGS}")
set(NCNN_ROOT "${CMAKE_SOURCE_DIR}/third_party" CACHE PATH "Directory containing ncnn-master and ncnn-build")
if(NOT EXISTS "${NCNN_ROOT}/ncnn-build/src/libncnn.a")
  message(FATAL_ERROR "Build NCNN first with scripts/build_tx2.sh, or set NCNN_ROOT to an existing build")
endif()
add_executable(detector src/person_tracker/detector.cpp)
add_executable(infer src/image_infer/infer.cpp)
foreach(target detector infer)
  target_include_directories(${target} PRIVATE ${OpenCV_INCLUDE_DIRS} ${NCNN_ROOT}/ncnn-master/src ${NCNN_ROOT}/ncnn-build/src)
  target_link_libraries(${target} ${NCNN_ROOT}/ncnn-build/src/libncnn.a ${OpenCV_LIBS} ${CMAKE_THREAD_LIBS_INIT})
endforeach()
